% recursive tail-call macro loop
\def\loopDown#1{%
  \testcounter=#1\relax
  \ifnum\testcounter>0
    [\the\testcounter]\ %
    \advance\testcounter by -1
    \expandafter\loopDown\expandafter{\the\testcounter}%
  \fi
}

% scope isolation test
\def\testScope{%
  \def\scopedVar{outer local}
  \global\def\globalVar{outer global}
  \begingroup
    \def\scopedVar{inner overridden local}
    \global\def\globalVar{inner overridden global}
  \endgroup
  scoped: \scopedVar\ | global: \globalVar
}

\def\contFrac#1#2#3{%
  #1 + \frac{1}{#2 + \frac{1}{#3 + \frac{1}{\dots}}}%
}
